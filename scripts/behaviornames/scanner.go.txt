// Command behaviornames extracts Go declarations for the behavior-name guard.
// It parses source without compiling it, so build tags and platform files are
// checked together. String payloads and comments are never declarations.
package main

import (
	"encoding/json"
	"fmt"
	"go/ast"
	"go/parser"
	"go/token"
	"os"
	"path/filepath"
	"sort"
	"strings"
)

// declaration records one exact name in its source-file category.
type declaration struct {
	// Path is the source path relative to the scanned root.
	Path string `json:"path"`
	// Symbol is the declared Go identifier.
	Symbol string `json:"symbol"`
	// Category distinguishes test entry points from other identifiers.
	Category string `json:"category"`
}

func main() {
	if len(os.Args) != 2 {
		fmt.Fprintln(os.Stderr, "usage: behaviornames ROOT")
		os.Exit(2)
	}
	root := os.Args[1]
	result := []declaration{}
	err := filepath.WalkDir(root, func(path string, d os.DirEntry, err error) error {
		if err != nil {
			return err
		}
		if d.IsDir() {
			if path != root && (d.Name() == ".git" || d.Name() == ".tmp") {
				return filepath.SkipDir
			}
			return nil
		}
		if !strings.HasSuffix(path, ".go") {
			return nil
		}
		if d.Type()&os.ModeSymlink != 0 {
			return fmt.Errorf("symlink input: %s", path)
		}
		f, err := parser.ParseFile(token.NewFileSet(), path, nil, parser.AllErrors)
		if err != nil {
			return err
		}
		rel, err := filepath.Rel(root, path)
		if err != nil {
			return err
		}
		seen := map[string]bool{}
		add := func(id *ast.Ident, category string) {
			if id == nil || id.Name == "_" {
				return
			}
			key := category + ":" + id.Name
			if !seen[key] {
				result = append(result, declaration{filepath.ToSlash(rel), id.Name, category})
				seen[key] = true
			}
		}
		add(f.Name, "identifier")
		ast.Inspect(f, func(n ast.Node) bool {
			switch node := n.(type) {
			case *ast.FuncDecl:
				category := "identifier"
				if node.Recv == nil {
					for _, prefix := range []string{"Test", "Benchmark", "Fuzz", "Example"} {
						if strings.HasPrefix(node.Name.Name, prefix) {
							category = "test"
							break
						}
					}
				}
				add(node.Name, category)
			case *ast.TypeSpec:
				add(node.Name, "identifier")
			case *ast.ValueSpec:
				for _, name := range node.Names {
					add(name, "identifier")
				}
			case *ast.Field:
				for _, name := range node.Names {
					add(name, "identifier")
				}
			case *ast.ImportSpec:
				add(node.Name, "identifier")
			case *ast.LabeledStmt:
				add(node.Label, "identifier")
			case *ast.Ident:
				if node.Obj != nil && node.Obj.Pos() == node.Pos() {
					// Functions are classified by their declaration above.
					if _, ok := node.Obj.Decl.(*ast.FuncDecl); !ok {
						add(node, "identifier")
					}
				}
			}
			return true
		})
		return nil
	})
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	sort.Slice(result, func(i, j int) bool {
		a, b := result[i], result[j]
		if a.Path != b.Path {
			return a.Path < b.Path
		}
		if a.Category != b.Category {
			return a.Category < b.Category
		}
		return a.Symbol < b.Symbol
	})
	if err := json.NewEncoder(os.Stdout).Encode(result); err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
}
